-- STEP 2: AUTOMATIC USER PROFILE CREATION
CREATE OR REPLACE FUNCTION public.handle_new_user()
RETURNS TRIGGER
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = public
AS $$
BEGIN
  INSERT INTO public.people (id, email)
  VALUES (new.id, new.email);
  RETURN new;
END;
$$;

-- Create a trigger that calls the function after a new user is created.
-- We drop it first to ensure this script can be re-run.
DROP TRIGGER IF EXISTS on_auth_user_created ON auth.users;
CREATE TRIGGER on_auth_user_created
  AFTER INSERT ON auth.users
  FOR EACH ROW EXECUTE PROCEDURE public.handle_new_user();


-- STEP 3: CREATE THE AUTH TOKEN HOOK FUNCTION (CORRECTED)
-- Uses SECURITY DEFINER to bypass RLS for its own internal queries.
CREATE OR REPLACE FUNCTION public.custom_access_token_hook(event json)
RETURNS json
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = public
AS $$
DECLARE
  claims jsonb;
  user_id uuid;
  org_id uuid;
  user_roles jsonb;
BEGIN
  user_id := (event->'token'->>'sub')::uuid;

  SELECT p.organization_id INTO org_id
  FROM public.people p
  WHERE p.id = user_id;

  SELECT COALESCE(jsonb_agg(r.name), '[]'::jsonb) INTO user_roles
  FROM public.user_roles ur
  JOIN public.roles r ON ur.role_id = r.id
  WHERE ur.person_id = user_id;

  claims := (event->'token'->'claims')::jsonb;

  claims := jsonb_set(
    claims,
    '{app_metadata}',
    jsonb_build_object(
      'organization_id', org_id,
      'roles', user_roles
    )
  );
  
  event := jsonb_set(event::jsonb, '{token,claims}', claims);

  RETURN event;
END;
$$;


-- STEP 4: CREATE A HELPER FUNCTION FOR RLS POLICIES (CORRECTED)
-- Correctly returns text which can be cast to uuid.
CREATE OR REPLACE FUNCTION public.get_claim(claim TEXT)
RETURNS text 
LANGUAGE sql STABLE
AS $$
  SELECT current_setting('request.jwt.claims', true)::jsonb -> 'app_metadata' ->> claim;
$$;


-- STEP 5: RECREATE ALL RLS POLICIES USING THE NEW JWT CLAIMS
CREATE OR REPLACE PROCEDURE public.create_org_rls_policy(table_name TEXT, check_column TEXT)
LANGUAGE plpgsql
AS $$
BEGIN
  EXECUTE format('ALTER TABLE public.%I ENABLE ROW LEVEL SECURITY;', table_name);
  EXECUTE format('DROP POLICY IF EXISTS "Allow members to access data in their own org" ON public.%I;', table_name);
  EXECUTE format(
    'CREATE POLICY "Allow members to access data in their own org" ON public.%I FOR ALL
     USING (%I = (public.get_claim(''organization_id''))::uuid)
     WITH CHECK (%I = (public.get_claim(''organization_id''))::uuid);',
    table_name, check_column, check_column
  );
END;
$$;
